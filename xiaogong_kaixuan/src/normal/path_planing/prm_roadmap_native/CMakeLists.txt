cmake_minimum_required(VERSION 3.16)
project(left_arm_prm_roadmap LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_planners_ompl REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(ompl REQUIRED)
add_executable(prm_roadmap_builder main.cpp)
ament_target_dependencies(prm_roadmap_builder rclcpp moveit_ros_planning moveit_planners_ompl moveit_msgs)
target_include_directories(prm_roadmap_builder PRIVATE ${OMPL_INCLUDE_DIRS})
target_link_libraries(prm_roadmap_builder ${OMPL_LIBRARIES})
